## 概要

ユーザーが好きなSCP番号を複数入力できるScpNumberInputコンポーネントを実装する。
入力バリデーション、追加・削除、API送信を管理。

## ユーザーストーリー

**ペルソナ**: SCPファン（熟練者）
**目的**: 好きなSCP番号を複数入力して自分だけの初期プロファイルを作る
**価値**: 自分の好みに合った精度の高い推薦を受けられる
**理由**: 既にお気に入りのSCPがあるので、それを基準に推薦してほしい

> SCPファンとして、好きなSCP番号を複数入力して自分だけの初期プロファイルを作りたい。なぜなら既にお気に入りのSCPがあるので、それを基準に推薦してほしいから。

      THEN エラーメッセージが表示される
      AND 追加されない

### AC-4: 重複チェック

- [x] WHEN ユーザーが既に追加済みの番号を入力した際
      THEN 「既に追加済みです」エラーが表示される
      AND 追加されない

### AC-5: SCP番号削除

- [x] WHEN ユーザーが入力済み番号の「×」をクリックした際
      THEN その番号がリストから削除される

### AC-6: 最低件数チェック

- [x] WHEN 入力済み番号が3件未満の際
      THEN 「始める」ボタンが無効になる
      AND 「あと○件必要です」メッセージが表示される

### AC-7: オンボーディング確定

- [x] WHEN ユーザーが「始める」ボタンをクリックした際
      GIVEN 3件以上の番号が入力されている場合
      THEN POST /onboarding/select/custom APIが呼び出される
      AND ローディング状態が表示される
      AND 成功時は記事閲覧画面に遷移する

### AC-8: 存在しないSCP番号エラー

- [x] WHEN POST /onboarding/select/custom APIが404を返した際
      GIVEN 存在しないSCP番号が含まれている場合
      THEN エラーメッセージに該当番号が表示される
      AND 該当番号がハイライト表示される

### AC-9: 戻るボタン

- [x] WHEN ユーザーが「戻る」をクリックした際
      THEN 選択肢画面に戻る
      AND 入力状態は破棄される

### AC-10: キーボード操作

- [x] WHEN 入力フィールドでEnterキーを押した際
      THEN 「追加」ボタンと同じ動作をする

## 技術設計

### 入力形式パターン

```typescript
// 許可される入力形式
const SCP_NUMBER_PATTERNS = [
  /^(\d{1,4})$/, // 例: 173, 2000
  /^SCP-(\d{1,4})$/i, // 例: SCP-173, scp-173
  /^SCP_(\d{1,4})$/i, // 例: SCP_173
];

// 正規化関数
function normalizeScpNumber(input: string): string | null {
  for (const pattern of SCP_NUMBER_PATTERNS) {
    const match = input.trim().match(pattern);
    if (match) {
      const num = match[1];
      return `scp-${num.padStart(3, "0")}`; // 例: "scp-173"
    }
  }
  return null;
}
```

interface UseScpNumberInputResult {
  /** 入力中の値 */
  inputValue: string;
  setInputValue: (value: string) => void;
  /** 追加済みSCP番号リスト（正規化済み） */
  scpNumbers: string[];
  /** 入力バリデーションエラー */
  inputError: string | null;
  /** 番号を追加 */
  addNumber: () => void;
  /** 番号を削除 */
  removeNumber: (number: string) => void;
  /** 最低件数を満たしているか */
  isValid: boolean;
  /** 残り必要件数 */
  remainingCount: number;
  /** オンボーディング確定 */
  confirmSelection: () => Promise<void>;
  /** 確定処理中 */
  isConfirming: boolean;
  /** 確定エラー */
  confirmError: Error | null;
  /** 存在しないSCP番号（APIエラー時） */
  invalidNumbers: string[];
}

function useScpNumberInput(visitorId: string): UseScpNumberInputResult;

### 実装方針

```tsx
// app/onboarding/_components/ScpNumberInput/index.tsx
"use client";

import { useScpNumberInput } from "./useScpNumberInput";

interface ScpNumberInputProps {
  visitorId: string;
  onComplete: () => void;
  onBack: () => void;
}

export function ScpNumberInput({ visitorId, onComplete, onBack }: ScpNumberInputProps) {
  const {
    inputValue,
    setInputValue,
    scpNumbers,
    inputError,
    addNumber,
    removeNumber,
    isValid,
    remainingCount,
    confirmSelection,
    isConfirming,
    confirmError,
    invalidNumbers,
  } = useScpNumberInput(visitorId);

  const handleSubmit = async () => {
    try {
      await confirmSelection();
      onComplete();
    } catch {
      // エラーはhook内で処理
    }
  };

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === "Enter") {
      e.preventDefault();
      addNumber();
    }
  };

  return (
    <div className="flex flex-col min-h-screen p-4">
      <header className="mb-6">
        <button onClick={onBack} className="text-gray-500">
          ← 戻る
        </button>
        <h1 className="text-xl font-bold mt-2">好きなSCPを教えてください</h1>
        <p className="text-sm text-gray-600 mt-1">最低3つのSCP番号を入力してください</p>
      </header>

      {/* 入力フォーム */}
      <div className="flex gap-2 mb-4">
        <input
          type="text"
          value={inputValue}
          onChange={(e) => setInputValue(e.target.value)}
          onKeyDown={handleKeyDown}
          placeholder="例: 173, SCP-682"
          className="flex-1 p-3 border rounded-lg"
          disabled={isConfirming}
        />
        <button
          onClick={addNumber}
          disabled={isConfirming || !inputValue.trim()}
          className="px-4 py-2 bg-gray-200 rounded-lg disabled:opacity-50"
        >
          追加
        </button>
      </div>

      {inputError && <p className="text-red-500 text-sm mb-4">{inputError}</p>}

      {/* 入力済みリスト */}
      <div className="flex-1">
        <div className="flex flex-wrap gap-2">
          {scpNumbers.map((num) => (
            <ScpNumberTag
              key={num}
              number={num}
              onRemove={() => removeNumber(num)}
              isInvalid={invalidNumbers.includes(num)}
              disabled={isConfirming}
            />
          ))}
        </div>

        {scpNumbers.length === 0 && (
          <p className="text-gray-400 text-center mt-8">まだSCP番号が入力されていません</p>
        )}
      </div>

      {/* 残り件数 */}
      {!isValid && (
        <p className="text-sm text-gray-500 text-center mt-4">あと{remainingCount}件必要です</p>
      )}

      {/* 確定エラー */}
      {confirmError && (
        <div className="text-red-500 text-sm mt-4">
          {invalidNumbers.length > 0
            ? `存在しないSCP番号があります: ${invalidNumbers.join(", ")}`
            : "エラーが発生しました。もう一度お試しください。"}
        </div>
      )}

      {/* 確定ボタン */}
      <footer className="mt-6">
        <button
          onClick={handleSubmit}
          disabled={!isValid || isConfirming}
          className="w-full py-3 bg-blue-600 text-white rounded-lg disabled:opacity-50"
        >
          {isConfirming ? "設定中..." : `始める（${scpNumbers.length}件選択中）`}
        </button>
      </footer>
    </div>
  );
}
```

### ScpNumberTag コンポーネント

```tsx
interface ScpNumberTagProps {
  number: string;
  onRemove: () => void;
  isInvalid: boolean;
  disabled: boolean;
}

function ScpNumberTag({ number, onRemove, isInvalid, disabled }: ScpNumberTagProps) {
  // "scp-173" → "SCP-173" に表示用変換
  const displayNumber = number.toUpperCase();

  return (
    <span
      className={`inline-flex items-center gap-1 px-3 py-1 rounded-full text-sm ${
        isInvalid ? "bg-red-100 text-red-800 border border-red-300" : "bg-gray-100 text-gray-800"
      }`}
    >
      {displayNumber}
      <button
        onClick={onRemove}
        disabled={disabled}
        className="ml-1 text-gray-500 hover:text-gray-700 disabled:opacity-50"
        aria-label={`${displayNumber}を削除`}
      >
        ×
      </button>
    </span>
  );
}
```
