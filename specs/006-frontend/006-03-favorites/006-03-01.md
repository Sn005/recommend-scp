**目的**: お気に入り画面にアクセスし、保存した記事を管理する

## 技術設計

### ページ構成

```typescript
// page.tsx
export default function FavoritesPage() {
  const { favorites, isLoading, error, removeFavorite, refresh } = useFavorites();

  if (isLoading) return <SkeletonLoader />;
  if (error) return <ErrorState onRetry={refresh} />;
  if (favorites.length === 0) return <EmptyState />;

  return (
    <>
      <MenuButton />
      <main className="pb-8 px-4">
        <Header count={favorites.length} />
        <FavoriteList favorites={favorites} onRemove={removeFavorite} />
      </main>
    </>
  );
}
```
