# Subtask 006-02-06: お気に入りボタン

## 概要

お気に入りのトグル状態管理とOptimistic UIを実装するuseArticleFavoriteフックを実装する。
状態の即時反映、APIロールバック、記事切り替え時の状態更新を担当。

## ステータス

- **status**: completed

## 親Story

[006-02: 記事閲覧](./006-02.md)

## ユーザーストーリー

**ペルソナ**: SCPファン
**目的**: 気に入った記事をお気に入りに保存する
**価値**: 後から読み返したい記事を簡単に見つけられる
**理由**: 好きな記事を忘れずに保管したい

> SCPファンとして、気に入った記事をお気に入りに保存し、後から読み返したい記事を簡単に見つけたい。なぜなら好きな記事を忘れずに保管したいから。

## Acceptance Criteria（EARS記法）

### AC-1: 初期状態表示

- [x] WHEN 記事が表示された際
      THEN お気に入りボタンが未選択状態（♡）で表示される
      AND 既にお気に入りの場合は選択状態（♥）で表示される

### AC-2: お気に入り追加

- [x] WHEN ユーザーがお気に入りボタンをタップした際
      GIVEN 記事がお気に入りに未追加の場合
      THEN POST /feedback APIでtype=favoriteを送信する
      AND ボタンが選択状態（♥）に即座に変化する（楽観的更新）
      AND ポップアニメーションが再生される

### AC-3: お気に入り解除

- [x] WHEN ユーザーがお気に入りボタンをタップした際
      GIVEN 記事が既にお気に入りの場合
      THEN DELETE /favorites/{articleId} APIで解除する
      AND ボタンが未選択状態（♡）に即座に変化する

### AC-4: Optimistic UI

- [x] WHEN お気に入りボタンがタップされた際
      THEN 即座にUIを更新する（楽観的更新）
      AND API失敗時はUIをロールバックする
      AND エラートーストを表示する

### AC-5: 記事切り替え時の状態リセット

- [x] WHEN 表示記事が切り替わった際
      THEN お気に入り状態を新しい記事のものに更新する
      AND ローカルキャッシュを参照する

### AC-6: 連打防止

- [x] WHEN お気に入りボタンが連続でタップされた際
      GIVEN 前回のAPI呼び出しが完了していない場合
      THE SYSTEM SHALL 追加のタップを無視する

### インターフェース定義

```typescript
// _hooks/useArticleFavorite.ts
interface UseArticleFavoriteOptions {
  /** 記事ID */
  articleId: string | undefined;
  /** 初期お気に入り状態 */
  initialFavorited?: boolean;
}

interface UseArticleFavoriteReturn {
  /** お気に入り状態 */
  isFavorited: boolean;
  /** 処理中フラグ */
  isProcessing: boolean;
  /** お気に入りをトグル */
  toggleFavorite: () => Promise<void>;
  /** お気に入りを追加 */
  addFavorite: () => Promise<void>;
  /** お気に入りを解除 */
  removeFavorite: () => Promise<void>;
}

export function useArticleFavorite(options: UseArticleFavoriteOptions): UseArticleFavoriteReturn;
```

### フック実装

```typescript
// _hooks/useArticleFavorite.ts
"use client";

import { useState, useCallback, useEffect, useRef } from "react";
import { api } from "@/shared/lib/api-client";
import { useFeedback } from "./useFeedback";

// ローカルキャッシュ（セッション中のお気に入り状態を保持）
const favoriteCache = new Map<string, boolean>();

export function useArticleFavorite({
  articleId,
  initialFavorited = false,
}: UseArticleFavoriteOptions): UseArticleFavoriteReturn {
  const [isFavorited, setIsFavorited] = useState(() => {
    if (!articleId) return false;
    return favoriteCache.get(articleId) ?? initialFavorited;
  });
  const [isProcessing, setIsProcessing] = useState(false);
  const { recordFavorite } = useFeedback();
  const abortControllerRef = useRef<AbortController | null>(null);

  // 記事切り替え時に状態を更新
  useEffect(() => {
    if (!articleId) {
      setIsFavorited(false);
      return;
    }

    const cachedState = favoriteCache.get(articleId);
    if (cachedState !== undefined) {
      setIsFavorited(cachedState);
    } else {
      setIsFavorited(initialFavorited);
    }
  }, [articleId, initialFavorited]);

  // お気に入り追加
  const addFavorite = useCallback(async () => {
    if (!articleId || isProcessing || isFavorited) return;

    // 進行中のリクエストをキャンセル
    abortControllerRef.current?.abort();
    abortControllerRef.current = new AbortController();

    // 楽観的更新
    setIsFavorited(true);
    favoriteCache.set(articleId, true);
    setIsProcessing(true);

    try {
      // フィードバック記録（favorite）
      await recordFavorite(articleId);
    } catch (error) {
      // ロールバック
      setIsFavorited(false);
      favoriteCache.set(articleId, false);
      // エラートースト表示（実装は別途）
      // NOTE: 本番実装ではcreateLoggerを使用すること
    } finally {
      setIsProcessing(false);
    }
  }, [articleId, isProcessing, isFavorited, recordFavorite]);

  // お気に入り解除
  const removeFavorite = useCallback(async () => {
    if (!articleId || isProcessing || !isFavorited) return;

    // 進行中のリクエストをキャンセル
    abortControllerRef.current?.abort();
    abortControllerRef.current = new AbortController();

    // 楽観的更新
    setIsFavorited(false);
    favoriteCache.set(articleId, false);
    setIsProcessing(true);

    try {
      await api.favorites[":articleId"].$delete({
        param: { articleId },
      });
    } catch (error) {
      // ロールバック
      setIsFavorited(true);
      favoriteCache.set(articleId, true);
      // エラートースト表示（実装は別途）
      // NOTE: 本番実装ではcreateLoggerを使用すること
    } finally {
      setIsProcessing(false);
    }
  }, [articleId, isProcessing, isFavorited]);

  // トグル
  const toggleFavorite = useCallback(async () => {
    if (isFavorited) {
      await removeFavorite();
    } else {
      await addFavorite();
    }
  }, [isFavorited, addFavorite, removeFavorite]);

  return {
    isFavorited,
    isProcessing,
    toggleFavorite,
    addFavorite,
    removeFavorite,
  };
}
```

### 親コンポーネントでの使用例

```tsx
function RecommendPage() {
  const { articles, currentIndex } = useInfiniteArticles();
  const currentArticle = articles[currentIndex];

  const { isFavorited, toggleFavorite, isProcessing } = useArticleFavorite({
    articleId: currentArticle?.id,
  });

  return (
    <FloatingUI
      isFavorited={isFavorited}
      onFavorite={toggleFavorite}
      // isProcessing中はボタンを無効化するオプションも検討
    />
  );
}
```
