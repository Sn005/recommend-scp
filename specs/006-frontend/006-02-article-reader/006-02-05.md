# Subtask 006-02-05: フィードバック記録

## 概要

Like/Dislike/FavoriteのAPI連携を管理するuseFeedbackフックを実装する。
読了判定（暗黙的Like）、スキップ（Dislike）、重複防止、オフライン対応を担当。

## ステータス

- **status**: completed

## ユーザーストーリー

**ペルソナ**: SCPファン
**目的**: 記事への反応（好き/スキップ）がシステムに記録される
**価値**: 自分の好みを学習してより良い推薦を受けられる
**理由**: 自分好みの記事をもっと読みたい

> SCPファンとして、記事への反応（好き/スキップ）がシステムに記録され、自分の好みを学習してより良い推薦を受けたい。なぜなら自分好みの記事をもっと読みたいから。

## Acceptance Criteria（EARS記法）

### AC-1: 読了判定（暗黙的Like）

- [x] WHEN 記事のスクロール率が90%に達した際
      GIVEN その記事で初めての場合
      THEN POST /feedback APIでtype=likeを送信する
      AND 記録済みフラグを設定する

### AC-2: スキップ（Dislike）

- [x] WHEN ユーザーがPillNavの「次へ」ボタンをタップした際
      GIVEN スクロール率が90%未満の場合
      THEN POST /feedback APIでtype=dislikeを送信する
      AND 次の記事に遷移する

### AC-3: 重複記録防止

- [x] WHERE フィードバック記録機能において
      IF 同じ記事に対して既にフィードバックが記録されている場合
      THE SYSTEM SHALL 追加のAPI呼び出しをスキップする

### AC-4: オフライン対応

- [x] WHEN フィードバックAPI呼び出しが失敗した際
      THEN ローカルにフィードバックをキューイングする
      AND ネットワーク復帰時に再送信する

### AC-5: エラーハンドリング

- [x] WHEN フィードバックAPI呼び出しでエラーが発生した際
      THEN ユーザー操作はブロックしない
      AND バックグラウンドでリトライする（最大3回）

### AC-6: フィードバック種別の優先度

- [x] WHERE 同じ記事に複数のフィードバックが発生した場合
      IF favorite > like > dislike の優先度で
      THE SYSTEM SHALL 最高優先度のフィードバックのみを送信する

### インターフェース定義

```typescript
// _hooks/useFeedback.ts
type FeedbackType = "like" | "dislike" | "favorite";

interface UseFeedbackReturn {
  /** 暗黙的Likeを記録（読了時） */
  recordLike: (articleId: string) => Promise<void>;
  /** Dislikeを記録（スキップ時） */
  recordDislike: (articleId: string) => Promise<void>;
  /** 明示的Favoriteを記録 */
  recordFavorite: (articleId: string) => Promise<void>;
  /** 記事のフィードバック済み状態を確認 */
  hasRecorded: (articleId: string) => boolean;
  /** フィードバック種別を取得 */
  getFeedbackType: (articleId: string) => FeedbackType | null;
  /** 保留中のフィードバック数 */
  pendingCount: number;
}

export function useFeedback(): UseFeedbackReturn;
```

### フック実装

```typescript
// _hooks/useFeedback.ts
"use client";

import { useState, useCallback, useEffect, useRef } from "react";
import { api } from "@/shared/lib/api-client";

type FeedbackType = "like" | "dislike" | "favorite";

interface PendingFeedback {
  articleId: string;
  type: FeedbackType;
  retryCount: number;
  timestamp: number;
}

const STORAGE_KEY = "scp-feedback-pending";
const MAX_RETRIES = 3;
const RETRY_DELAY_MS = 5000;

// 優先度: favorite > like > dislike
const PRIORITY: Record<FeedbackType, number> = {
  favorite: 3,
  like: 2,
  dislike: 1,
};

export function useFeedback(): UseFeedbackReturn {
  const [recordedFeedbacks, setRecordedFeedbacks] = useState<Map<string, FeedbackType>>(new Map());
  const [pendingQueue, setPendingQueue] = useState<PendingFeedback[]>([]);
  const processingRef = useRef(false);

  // 保留中のフィードバックをローカルストレージから復元
  useEffect(() => {
    const stored = localStorage.getItem(STORAGE_KEY);
    if (stored) {
      try {
        const pending = JSON.parse(stored) as PendingFeedback[];
        setPendingQueue(pending);
      } catch {
        // 無視
      }
    }
  }, []);

  // 保留中のフィードバックをローカルストレージに保存
  useEffect(() => {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(pendingQueue));
  }, [pendingQueue]);

  // フィードバック送信（内部）
  const sendFeedback = useCallback(
    async (articleId: string, type: FeedbackType): Promise<boolean> => {
      try {
        await api.feedback.$post({
          json: { articleId, type },
        });
        return true;
      } catch {
        return false;
      }
    },
    []
  );

  // キュー処理
  const processQueue = useCallback(async () => {
    if (processingRef.current || pendingQueue.length === 0) return;

    processingRef.current = true;

    const [current, ...rest] = pendingQueue;
    const success = await sendFeedback(current.articleId, current.type);

    if (success) {
      // 成功: キューから削除、記録済みに追加
      setPendingQueue(rest);
      setRecordedFeedbacks((prev) => new Map(prev).set(current.articleId, current.type));
    } else if (current.retryCount < MAX_RETRIES) {
      // リトライ: カウントを増やしてキューの末尾へ
      setPendingQueue([...rest, { ...current, retryCount: current.retryCount + 1 }]);
    } else {
      // 最大リトライ超過: キューから削除
      setPendingQueue(rest);
    }

    processingRef.current = false;
  }, [pendingQueue, sendFeedback]);

  // 定期的にキューを処理
  useEffect(() => {
    const interval = setInterval(processQueue, RETRY_DELAY_MS);
    return () => clearInterval(interval);
  }, [processQueue]);

  // オンライン復帰時にキューを処理
  useEffect(() => {
    const handleOnline = () => processQueue();
    window.addEventListener("online", handleOnline);
    return () => window.removeEventListener("online", handleOnline);
  }, [processQueue]);

  // フィードバック記録（共通）
  const recordFeedback = useCallback(
    async (articleId: string, type: FeedbackType) => {
      // 既存のフィードバックと優先度を比較
      const existing = recordedFeedbacks.get(articleId);
      if (existing && PRIORITY[existing] >= PRIORITY[type]) {
        // 既存の方が優先度が高い場合はスキップ
        return;
      }

      // 即座に記録済みとしてマーク（楽観的更新）
      setRecordedFeedbacks((prev) => new Map(prev).set(articleId, type));

      // API送信を試行
      const success = await sendFeedback(articleId, type);

      if (!success) {
        // 失敗時はキューに追加
        setPendingQueue((prev) => [
          ...prev.filter((p) => p.articleId !== articleId),
          { articleId, type, retryCount: 0, timestamp: Date.now() },
        ]);
      }
    },
    [recordedFeedbacks, sendFeedback]
  );

  // 各フィードバック種別のラッパー
  const recordLike = useCallback(
    (articleId: string) => recordFeedback(articleId, "like"),
    [recordFeedback]
  );

  const recordDislike = useCallback(
    (articleId: string) => recordFeedback(articleId, "dislike"),
    [recordFeedback]
  );

  const recordFavorite = useCallback(
    (articleId: string) => recordFeedback(articleId, "favorite"),
    [recordFeedback]
  );

  const hasRecorded = useCallback(
    (articleId: string) => recordedFeedbacks.has(articleId),
    [recordedFeedbacks]
  );

  const getFeedbackType = useCallback(
    (articleId: string) => recordedFeedbacks.get(articleId) ?? null,
    [recordedFeedbacks]
  );

  return {
    recordLike,
    recordDislike,
    recordFavorite,
    hasRecorded,
    getFeedbackType,
    pendingCount: pendingQueue.length,
  };
}
```

### 親コンポーネントでの使用例

```tsx
function RecommendPage() {
  const { recordLike, recordDislike, hasRecorded } = useFeedback();
  const { articles, currentIndex, goToNext } = useInfiniteArticles();

  const currentArticle = articles[currentIndex];

  const handleScrollEnd = () => {
    // 読了時に暗黙的Like
    if (!hasRecorded(currentArticle.id)) {
      recordLike(currentArticle.id);
    }
  };

  const handleNext = () => {
    // スキップ時にDislike（未記録の場合のみ）
    if (!hasRecorded(currentArticle.id)) {
      recordDislike(currentArticle.id);
    }
    goToNext();
  };

  return <ArticleWebView url={currentArticle.url} onScrollEnd={handleScrollEnd} />;
}
```
