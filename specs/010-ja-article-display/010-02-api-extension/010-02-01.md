### Repository変更

```typescript
// packages/shared/src/recommendation/article-repository.ts
export const getRecommendableArticles = async (
  supabase: SupabaseClient,
  options: { lang?: string } = {}
) => {
  const { lang = "ja" } = options;

  const { data, error } = await supabase
    .from("scp_articles")
    .select(
      `
      article_id,
      title,
      tags,
      embedding,
      article_translations!inner (
        url,
        has_translation
      )
    `
    )
    .eq("article_translations.lang", lang)
    .or("has_translation.is.null,has_translation.eq.true", {
      foreignTable: "article_translations",
    });

  return data;
};
```

### 推薦エンジン変更

```typescript
// packages/shared/src/recommendation/engine.ts
const mapToRecommendedArticle = (
  article: ArticleWithTranslation,
  score: number,
  source: "preference" | "serendipity"
): RecommendedArticle => ({
  id: article.article_id,
  title: article.title,
  similarityScore: score,
  source,
  url: article.article_translations[0]?.url ?? "",
});
```
