## 技術設計

### repository.ts

```typescript
import type { SupabaseClient } from "@supabase/supabase-js";
import type { Feedback } from "@recommend-scp/shared";

export class FeedbackRepository {
  constructor(private supabase: SupabaseClient) {}

  async save(feedback: Omit<Feedback, "id">): Promise<Feedback> {
    const { data, error } = await this.supabase
      .from("feedback")
      .upsert(
        {
          visitor_id: feedback.visitorId,
          article_id: feedback.articleId,
          type: feedback.type,
          created_at: feedback.createdAt,
        },
        {
          onConflict: "visitor_id,article_id",
        }
      )
      .select()
      .single();

    if (error) throw error;

    return {
      id: data.id,
      visitorId: data.visitor_id,
      articleId: data.article_id,
      type: data.type,
      createdAt: data.created_at,
    };
  }

  async getByVisitorId(visitorId: string): Promise<Feedback[]> {
    const { data, error } = await this.supabase
      .from("feedback")
      .select("*")
      .eq("visitor_id", visitorId);

    if (error) throw error;

    return data.map((row) => ({
      id: row.id,
      visitorId: row.visitor_id,
      articleId: row.article_id,
      type: row.type,
      createdAt: row.created_at,
    }));
  }
}
```

### service.ts

```typescript
import { FeedbackRepository } from "./repository";
import { VisitorsRepository } from "../visitors/repository";
import { NotFoundError } from "../../lib/errors";

export class FeedbackService {
  constructor(
    private feedbackRepo: FeedbackRepository,
    private visitorsRepo: VisitorsRepository
  ) {}

  async recordFeedback(
    visitorId: string,
    articleId: string,
    type: "like" | "dislike"
  ): Promise<void> {
    // visitorIdの存在確認
    const visitor = await this.visitorsRepo.findByVisitorId(visitorId);
    if (!visitor) {
      throw new NotFoundError("Visitor", visitorId);
    }

    // フィードバックを保存
    await this.feedbackRepo.save({
      visitorId,
      articleId,
      type,
      createdAt: new Date().toISOString(),
    });

    // 嗜好ベクトル再計算は次回推薦時に実行（遅延評価）
  }
}
```
