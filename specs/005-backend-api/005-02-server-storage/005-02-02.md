# Subtask 005-02-02: SupabasePreferenceStorage

## 概要

EPIC-004で定義した `PreferenceStorage` インターフェースのSupabase実装を作成する。

## ステータス

- **status**: planned

## ユーザーストーリー

**As a** API開発者
**I want** PreferenceStorageのサーバー側実装を持つ
**So that** RecommendationEngineをサーバー側で動作させられる

## Acceptance Criteria（EARS記法）

- [ ] WHERE SupabasePreferenceStorage
      THE SYSTEM SHALL PreferenceStorageインターフェースを完全に実装する

- [ ] WHEN getProfile を呼び出した際
      GIVEN visitorIdが存在する場合
      THEN visitorsテーブルからプロファイルを取得する
      AND snake_case → camelCase 変換を行う

- [ ] WHEN saveProfile を呼び出した際
      THEN visitorsテーブルにupsertする
      AND camelCase → snake_case 変換を行う

- [ ] WHEN getViewHistory を呼び出した際
      THEN view_historyテーブルから履歴を取得する
      AND viewed_at降順でソートする

- [ ] WHEN addFeedback を呼び出した際
      GIVEN 同じvisitorId + articleIdの既存フィードバックがある場合
      THEN 既存のフィードバックを上書きする（UPSERT）

- [ ] WHEN getDislikedArticleIds を呼び出した際
      THEN type='dislike' のarticle_idリストを返す

- [ ] WHEN getFavorites を呼び出した際
      THEN favoritesテーブルからadded_at降順で取得する

### 実装

```typescript
import type { SupabaseClient } from "@supabase/supabase-js";
import type {
  PreferenceStorage,
  PreferenceProfile,
  ViewHistory,
  Feedback,
  RecommendationLog,
  Favorite,
} from "@recommend-scp/shared";

export class SupabasePreferenceStorage implements PreferenceStorage {
  constructor(private readonly supabase: SupabaseClient) {}

  async getProfile(visitorId: string): Promise<PreferenceProfile | null> {
    const { data, error } = await this.supabase
      .from("visitors")
      .select("*")
      .eq("visitor_id", visitorId)
      .single();

    if (error || !data) return null;
    return this.toPreferenceProfile(data);
  }

  async saveProfile(profile: PreferenceProfile): Promise<void> {
    const { error } = await this.supabase.from("visitors").upsert(this.toVisitorRow(profile), {
      onConflict: "visitor_id",
    });

    if (error) throw error;
  }

  async getViewHistory(visitorId: string, limit?: number): Promise<ViewHistory[]> {
    let query = this.supabase
      .from("view_history")
      .select("*")
      .eq("visitor_id", visitorId)
      .order("viewed_at", { ascending: false });

    if (limit) query = query.limit(limit);

    const { data, error } = await query;
    if (error) throw error;
    return (data ?? []).map(this.toViewHistory);
  }

  async addViewHistory(history: ViewHistory): Promise<void> {
    const { error } = await this.supabase.from("view_history").insert({
      visitor_id: history.visitorId,
      article_id: history.articleId,
      viewed_at: history.viewedAt,
      duration: history.duration,
    });
    if (error) throw error;
  }

  async getFeedback(visitorId: string): Promise<Feedback[]> {
    const { data, error } = await this.supabase
      .from("feedback")
      .select("*")
      .eq("visitor_id", visitorId);

    if (error) throw error;
    return (data ?? []).map(this.toFeedback);
  }

  async getFeedbackByArticle(visitorId: string, articleId: string): Promise<Feedback | null> {
    const { data, error } = await this.supabase
      .from("feedback")
      .select("*")
      .eq("visitor_id", visitorId)
      .eq("article_id", articleId)
      .single();

    if (error || !data) return null;
    return this.toFeedback(data);
  }

  async addFeedback(feedback: Feedback): Promise<void> {
    const { error } = await this.supabase.from("feedback").upsert(
      {
        visitor_id: feedback.visitorId,
        article_id: feedback.articleId,
        type: feedback.type,
        created_at: feedback.createdAt,
      },
      { onConflict: "visitor_id,article_id" }
    );
    if (error) throw error;
  }

  async getRecommendationLog(visitorId: string, limit?: number): Promise<RecommendationLog[]> {
    let query = this.supabase
      .from("recommendation_log")
      .select("*")
      .eq("visitor_id", visitorId)
      .order("recommended_at", { ascending: false });

    if (limit) query = query.limit(limit);

    const { data, error } = await query;
    if (error) throw error;
    return (data ?? []).map(this.toRecommendationLog);
  }

  async addRecommendationLog(log: RecommendationLog): Promise<void> {
    const { error } = await this.supabase.from("recommendation_log").insert({
      visitor_id: log.visitorId,
      article_id: log.articleId,
      source: log.source,
      recommended_at: log.recommendedAt,
      clicked: log.clicked,
    });
    if (error) throw error;
  }

  async getDislikedArticleIds(visitorId: string): Promise<string[]> {
    const { data, error } = await this.supabase
      .from("feedback")
      .select("article_id")
      .eq("visitor_id", visitorId)
      .eq("type", "dislike");

    if (error) throw error;
    return (data ?? []).map((d) => d.article_id);
  }

  async getArticleTags(articleId: string): Promise<string[] | null> {
    const { data, error } = await this.supabase
      .from("scp_articles")
      .select("tags")
      .eq("id", articleId)
      .single();

    if (error || !data) return null;
    return data.tags ?? [];
  }

  async getFavorites(visitorId: string): Promise<Favorite[]> {
    const { data, error } = await this.supabase
      .from("favorites")
      .select("*")
      .eq("visitor_id", visitorId)
      .order("added_at", { ascending: false });

    if (error) throw error;
    return (data ?? []).map(this.toFavorite);
  }

  async addFavorite(favorite: Favorite): Promise<void> {
    const { error } = await this.supabase.from("favorites").upsert(
      {
        visitor_id: favorite.visitorId,
        article_id: favorite.articleId,
        added_at: favorite.addedAt,
      },
      { onConflict: "visitor_id,article_id" }
    );
    if (error) throw error;
  }

  async removeFavorite(visitorId: string, articleId: string): Promise<void> {
    const { error } = await this.supabase
      .from("favorites")
      .delete()
      .eq("visitor_id", visitorId)
      .eq("article_id", articleId);
    if (error) throw error;
  }

  // Private: DB row → Domain model 変換
  private toPreferenceProfile(row: Record<string, unknown>): PreferenceProfile {
    return {
      visitorId: row.visitor_id as string,
      tagWeights: (row.tag_weights as Record<string, number>) ?? {},
      objectClassPreference: (row.object_class_preference as Record<string, number>) ?? {},
      starterPack: row.starter_pack as PreferenceProfile["starterPack"],
      onboardingCompletedAt: row.onboarding_completed_at as string | undefined,
      preferenceEmbedding: row.preference_vector as number[] | undefined,
      createdAt: row.created_at as string,
      updatedAt: row.updated_at as string,
    };
  }

  private toVisitorRow(profile: PreferenceProfile): Record<string, unknown> {
    return {
      visitor_id: profile.visitorId,
      tag_weights: profile.tagWeights,
      object_class_preference: profile.objectClassPreference,
      starter_pack: profile.starterPack,
      onboarding_completed_at: profile.onboardingCompletedAt,
      preference_vector: profile.preferenceEmbedding,
      updated_at: new Date().toISOString(),
    };
  }

  // ... 他の変換メソッド
}
```
