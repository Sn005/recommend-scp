### 実装

```typescript
import type { SupabaseClient } from "@supabase/supabase-js";
import type {
  VectorSearchClient,
  VectorSearchParams,
  VectorSearchResult,
  UnexploredTagsSearchParams,
} from "@recommend-scp/shared";

export class SupabaseVectorSearch implements VectorSearchClient {
  constructor(private readonly supabase: SupabaseClient) {}

  async searchByEmbedding(params: VectorSearchParams): Promise<VectorSearchResult[]> {
    const { data, error } = await this.supabase.rpc("search_articles_by_embedding", {
      query_vector: params.queryVector,
      exclude_ids: params.excludeIds ?? [],
      match_count: params.limit,
      min_similarity: params.minSimilarity ?? 0,
      max_similarity: params.maxSimilarity ?? 1,
    });

    if (error) throw error;

    return (data ?? []).map((row: Record<string, unknown>) => ({
      id: row.id as string,
      title: row.title as string,
      similarity: row.similarity as number,
    }));
  }

  async getEmbedding(articleId: string): Promise<number[] | null> {
    const { data, error } = await this.supabase
      .from("scp_articles")
      .select("embedding")
      .eq("id", articleId)
      .single();

    if (error || !data) return null;
    return data.embedding as number[] | null;
  }

  async searchByUnexploredTags(params: UnexploredTagsSearchParams): Promise<VectorSearchResult[]> {
    const { data, error } = await this.supabase.rpc("search_articles_by_unexplored_tags", {
      explored_tags: params.exploredTags,
      exclude_ids: params.excludeIds ?? [],
      match_count: params.limit,
      order_by: params.orderBy,
    });

    if (error) throw error;

    return (data ?? []).map((row: Record<string, unknown>) => ({
      id: row.id as string,
      title: row.title as string,
      similarity: 0.5, // 未探索タグ検索では固定値
    }));
  }
}
```
