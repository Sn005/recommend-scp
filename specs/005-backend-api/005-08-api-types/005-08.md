### 使用例

```typescript
// apps/web/lib/api-client.ts
import { hc } from "hono/client";
import type { AppType } from "@recommend-scp/api-types";

export const api = hc<AppType>(process.env.NEXT_PUBLIC_API_URL!);

// 型安全なAPI呼び出し
const result = await api.recommend.$post({
  json: {
    visitorId: "...",
    limit: 10,
  },
});
```
