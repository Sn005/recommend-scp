## 技術設計

### apps/api-server/src/app.ts

```typescript
import { Hono } from "hono";
import { createVisitorsRoutes } from "./domains/visitors/routes";
import { createArticlesRoutes } from "./domains/articles/routes";
import { createRecommendRoutes } from "./domains/recommend/routes";
import { createFeedbackRoutes } from "./domains/feedback/routes";
import { createOnboardingRoutes } from "./domains/onboarding/routes";
import { health } from "./routes/health";

export const createApp = (supabase: SupabaseClient) => {
  const app = new Hono()
    .route("/health", health)
    .route("/visitors", createVisitorsRoutes(supabase))
    .route("/articles", createArticlesRoutes(supabase))
    .route("/recommend", createRecommendRoutes(supabase))
    .route("/feedback", createFeedbackRoutes(supabase))
    .route("/onboarding", createOnboardingRoutes(supabase));

  return app;
};

// 型をexport（RPC用）
export type AppType = ReturnType<typeof createApp>;
```

### packages/api-types/src/index.ts

```typescript
// apps/api-serverから型を再export
export type { AppType } from "@recommend-scp/api-server";
```

### フロントエンドでの使用例

```typescript
// apps/web/lib/api-client.ts
import { hc } from "hono/client";
import type { AppType } from "@recommend-scp/api-types";

const client = hc<AppType>(process.env.NEXT_PUBLIC_API_URL!);

// 型安全なAPI呼び出し
export const api = {
  visitors: {
    register: (visitorId: string) => client.visitors.$post({ json: { visitorId } }),
  },
  recommend: {
    get: (visitorId: string, limit?: number) =>
      client.recommend.$post({ json: { visitorId, limit } }),
  },
  feedback: {
    record: (visitorId: string, articleId: string, type: "like" | "dislike") =>
      client.feedback.$post({ json: { visitorId, articleId, type } }),
  },
  onboarding: {
    getPacks: () => client.onboarding.packs.$get(),
    selectPack: (visitorId: string, packType: string) =>
      client.onboarding.select.$post({ json: { visitorId, packType } }),
  },
  articles: {
    search: (q: string, limit?: number) => client.articles.search.$get({ query: { q, limit } }),
  },
};
```
