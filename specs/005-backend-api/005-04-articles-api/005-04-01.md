## 技術設計

### repository.ts

```typescript
import type { SupabaseClient } from "@supabase/supabase-js";
import type { VectorSearchResult } from "@recommend-scp/shared";

export interface ArticleDetail {
  id: string;
  title: string;
  url: string;
  tags: string[];
  rating: number | null;
}

export class ArticlesRepository {
  constructor(private readonly supabase: SupabaseClient) {}

  async searchByEmbedding(
    queryVector: number[],
    options: { limit?: number; excludeIds?: string[] } = {}
  ): Promise<VectorSearchResult[]> {
    const { data, error } = await this.supabase.rpc("search_articles_by_embedding", {
      query_vector: queryVector,
      exclude_ids: options.excludeIds ?? [],
      match_count: options.limit ?? 10,
    });

    if (error) throw error;
    return data ?? [];
  }

  async getArticleById(id: string): Promise<ArticleDetail | null> {
    const { data, error } = await this.supabase
      .from("scp_articles")
      .select("id, title, url, tags, rating")
      .eq("id", id)
      .single();

    if (error?.code === "PGRST116") return null;
    if (error) throw error;
    return data;
  }
}
```

### service.ts

```typescript
import { ArticlesRepository } from "./repository";
import { createEmbedding } from "../../lib/openai";

export interface SearchArticlesResult {
  articles: Array<{
    id: string;
    title: string;
    similarity: number;
    tags?: string[];
  }>;
  total: number;
  query: string;
}

export class ArticlesService {
  constructor(private readonly repository: ArticlesRepository) {}

  async searchArticles(
    query: string,
    options: { limit?: number } = {}
  ): Promise<SearchArticlesResult> {
    // クエリをEmbeddingに変換
    const queryVector = await createEmbedding(query);

    // ベクトル検索
    const results = await this.repository.searchByEmbedding(queryVector, {
      limit: options.limit ?? 10,
    });

    return {
      articles: results.map((r) => ({
        id: r.id,
        title: r.title,
        similarity: r.similarity,
      })),
      total: results.length,
      query,
    };
  }
}
```

### lib/openai.ts

```typescript
import OpenAI from "openai";
import { env } from "@recommend-scp/shared";

const openai = new OpenAI({ apiKey: env.OPENAI_API_KEY });

export async function createEmbedding(text: string): Promise<number[]> {
  const response = await openai.embeddings.create({
    model: "text-embedding-3-small",
    input: text,
  });
  return response.data[0].embedding;
}
```
