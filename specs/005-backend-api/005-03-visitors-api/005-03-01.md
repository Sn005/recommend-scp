## 技術設計

### repository.ts

```typescript
import type { SupabaseClient } from "@supabase/supabase-js";

export interface VisitorRow {
  id: string;
  visitor_id: string;
  created_at: string;
  updated_at: string;
}

export class VisitorsRepository {
  constructor(private readonly supabase: SupabaseClient) {}

  async findByVisitorId(visitorId: string): Promise<VisitorRow | null> {
    const { data, error } = await this.supabase
      .from("visitors")
      .select("id, visitor_id, created_at, updated_at")
      .eq("visitor_id", visitorId)
      .single();

    if (error?.code === "PGRST116") return null; // Not found
    if (error) throw error;
    return data;
  }

  async create(visitorId: string): Promise<VisitorRow> {
    const { data, error } = await this.supabase
      .from("visitors")
      .insert({ visitor_id: visitorId })
      .select("id, visitor_id, created_at, updated_at")
      .single();

    if (error) throw error;
    return data;
  }

  async existsByVisitorId(visitorId: string): Promise<boolean> {
    const { count, error } = await this.supabase
      .from("visitors")
      .select("*", { count: "exact", head: true })
      .eq("visitor_id", visitorId);

    if (error) throw error;
    return (count ?? 0) > 0;
  }
}
```

### service.ts

```typescript
import { VisitorsRepository } from "./repository";

export interface RegisterVisitorResult {
  visitorId: string;
  isNew: boolean;
  createdAt: string;
}

export class VisitorsService {
  constructor(private readonly repository: VisitorsRepository) {}

  async registerVisitor(visitorId: string): Promise<RegisterVisitorResult> {
    const existing = await this.repository.findByVisitorId(visitorId);

    if (existing) {
      return {
        visitorId: existing.visitor_id,
        isNew: false,
        createdAt: existing.created_at,
      };
    }

    const created = await this.repository.create(visitorId);
    return {
      visitorId: created.visitor_id,
      isNew: true,
      createdAt: created.created_at,
    };
  }
}
```
